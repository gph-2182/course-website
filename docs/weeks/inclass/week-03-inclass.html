Exercise 1: Hospital ICU Capacity Matrix (Cycle 1)

A municipal health department monitors daily ICU admissions across three regional hospital sites.

# 1. Create a 3x3 matrix named icu_admissions from the vector below, filled row-by-row
raw_counts <- c(12, 18, 5, 24, 30, 11, 8, 15, 4)
icu_admissions <- matrix(raw_counts, nrow = 3, ncol = 3, byrow = TRUE)

# 2. Assign rownames: "Mon", "Tue", "Wed" and colnames: "Site_1", "Site_2", "Site_3"
rownames(icu_admissions) <- c("Mon", "Tue", "Wed")
colnames(icu_admissions) <- c("Site_1", "Site_2", "Site_3")
icu_admissions
    Site_1 Site_2 Site_3
Mon     12     18      5
Tue     24     30     11
Wed      8     15      4
# 3. Calculate total admissions for each hospital site across all three days
site_totals <- colSums(icu_admissions)
site_totals
Site_1 Site_2 Site_3 
    44     63     20 
# 4. Extract admissions for Tuesday across all sites as a named vector
tue_vitals <- icu_admissions["Tue", ]
tue_vitals
Site_1 Site_2 Site_3 
    24     30     11 
# 5. Extract all entries in the matrix where daily admissions were >= 20
high_admission_days <- icu_admissions[icu_admissions >= 20]
high_admission_days

Exercise 2: Clinical Trial Patient Record List (Cycle 2)

# 1. Construct a clinical trial patient record list named patient_profile
patient_profile <- list(
  subject_id = "SUBJ-1049",
  treatment_arm = "High_Dose",
  age = 58,
  viral_loads = c(45000, 12000, 850, 40),
  side_effects = c("Headache", "Fatigue"),
  site_info = list(city = "New York", hospital = "Bellevue")
)

# 2. Extract the viral_loads vector using [[ ]] and compute the percentage reduction:
#    ((baseline - final) / baseline) * 100
baseline_vl <- patient_profile[["viral_loads"]][1]
final_vl <- patient_profile[["viral_loads"]][4]
pct_reduction <- ((baseline_vl - final_vl) / baseline_vl) * 100

Exercise 3: Patient Cohort Data Frame & Subsetting (Cycle 3)

# Here is a sample cohort of cardiology patients
cohort_df <- data.frame(
  patient_id = c("P01", "P02", "P03", "P04", "P05", "P06"),
  age = c(45, 62, 38, 71, 55, 68),
  sbp = c(120, 145, 118, 160, 135, 150),
  dbp = c(80, 92, 76, 98, 88, 95),
  bmi = c(24.2, 31.0, 27.5, 29.8, 22.1, 33.4),
  smoker = c(FALSE, TRUE, FALSE, TRUE, FALSE, TRUE),
  stringsAsFactors = FALSE
)

# 1. Add a new column pulse_pressure = sbp - dbp
cohort_df$pulse_pressure <- cohort_df$sbp - cohort_df$dbp

# 2. Add a boolean column hypertensive = (sbp >= 140 | dbp >= 90)
cohort_df$hypertensive <- cohort_df$sbp >= 140 | cohort_df$dbp >= 90

# 3. Filter all patients who are hypertensive AND smokers
high_risk <- cohort_df[cohort_df$hypertensive & cohort_df$smoker, ]
high_risk
  patient_id age sbp dbp  bmi smoker pulse_pressure hypertensive
2        P02  62 145  92 31.0   TRUE             53         TRUE
4        P04  71 160  98 29.8   TRUE             62         TRUE
6        P06  68 150  95 33.4   TRUE             55         TRUE
# 4. Compute the average BMI of hypertensive patients vs non-hypertensive patients
mean(cohort_df$bmi[cohort_df$hypertensive])
[1] 31.4
mean(cohort_df$bmi[!cohort_df$hypertensive])
[1] 24.6
